# Criar o projeto Next.js

pnpm create next-app@latest portfolio-vitrine --js --tailwind --eslint

# Entre no diretório

cd portfolio-vitrine

# Inicializar Shadcn UI (escolha: Style: New York, Color: Zinc, CSS Variables: Yes)

npx shadcn-ui@latest init

pnpm add next-themes lucide-react

pnpm add next-themes@latest

npx shadcn-ui@latest add card badge button

# folder structure

```text
src/
├── app/
│   ├── layout.js       # Layout principal com Providers
│   ├── page.js         # Página da vitrine
│   └── globals.css
├── components/
│   ├── ui/             # Componentes do Shadcn
│   ├── theme-provider.js
│   ├── theme-toggle.js
│   └── project-card.js # (Próxima etapa)
├── data/
│   └── dadosProjetos.js # Fonte da verdade dos projetos
└── lib/
    └── utils.js        # Utilitários (clsx, tailwind-merge)
```
